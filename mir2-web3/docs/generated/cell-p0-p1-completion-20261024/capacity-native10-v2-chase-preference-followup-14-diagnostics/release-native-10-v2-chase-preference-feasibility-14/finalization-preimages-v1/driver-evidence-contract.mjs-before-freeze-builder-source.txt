import fs from 'node:fs/promises';import path from 'node:path';import assert from 'node:assert/strict';import {base,read,sha,samePath}from'./input-lib.mjs';import {DRIVER_BINDING,assertFinalDriverBinding}from'./driver-binding-contract.mjs';
export async function verifyDriverEvidenceBasis(frozen,assertPin){
 assertFinalDriverBinding();assert.equal(frozen.actualCodeCommit,DRIVER_BINDING.driverSourceCommit);assert.equal(frozen.driverSourceCommit,DRIVER_BINDING.driverSourceCommit);assert.equal(frozen.files?.length,9);assert.equal(frozen.sourceSha256,DRIVER_BINDING.driverSourceSha256);assert.equal(sha(JSON.stringify(frozen.files)),frozen.sourceSha256);assert.equal(frozen.capacityAccepted,false);assert.equal(frozen.otherSevenInputsUnchanged,true);
 const pins=[],seen=new Set(),add=async p=>{assert.ok(p&&typeof p.file==='string'&&path.isAbsolute(p.file));const real=await fs.realpath(p.file),rel=path.relative(await fs.realpath(base),real);assert.ok(rel&&!path.isAbsolute(rel)&&!rel.split(/[\\/]/).some(x=>x==='..'||/^(private|store|recordings|bin)$/i.test(x))&&samePath(real,p.file));await assertPin(p);if(!seen.has(p.file)){seen.add(p.file);pins.push(p);}};
 const expected=[{tests:13,receipt:frozen.focused.receipt,log:frozen.focused.log},{tests:200,receipt:frozen.fullRegression.receipt,log:frozen.fullRegression.log}];assert.deepEqual(DRIVER_BINDING.currentEvidence,expected);
 const sameSource=r=>{for(const s of[r.sourceBefore,r.sourceAfter]){assert.equal(s.sourceSha256,frozen.sourceSha256);assert.equal(sha(JSON.stringify(s.files)),s.sourceSha256);assert.deepEqual(s.files,frozen.files);}assert.deepEqual(r.sourceBefore,r.sourceAfter);};
 const tap=async(p,count)=>{await add(p);const text=await fs.readFile(p.file,'utf8');for(const[k,n]of[['tests',count],['pass',count],['fail',0],['cancelled',0],['skipped',0]])assert.ok(new RegExp('^# '+k+' '+n+'\\r?$','m').test(text),'Actual TAPfooter '+k);};
 await add(frozen.focused.receipt);const focused=await read(frozen.focused.receipt.file);sameSource(focused);assert.equal(focused.currentSourceRestored,true);assert.equal(focused.redExpectedBehavioralFailure,true);assert.equal(focused.red.exitCode,1);assert.deepEqual(focused.red.counts,{tests:13,pass:4,fail:9,cancelled:0,skipped:0});assert.equal(focused.green.exitCode,0);assert.equal(focused.green.signal,null);assert.equal(focused.green.error,null);assert.deepEqual(focused.green.counts,{tests:13,pass:13,fail:0,cancelled:0,skipped:0});assert.deepEqual(focused.green.log,frozen.focused.log);await add(focused.red.log);await tap(focused.green.log,13);
 await add(frozen.fullRegression.receipt);assert.deepEqual(frozen.fullRegression.exactSourceManifest,frozen.fullRegression.receipt);const full=await read(frozen.fullRegression.receipt.file);sameSource(full);assert.equal(full.exitCode,0);assert.equal(full.signal,null);assert.equal(full.error,null);assert.equal(full.sourcePinsUnchangedAfterRun,true);assert.deepEqual(full.counts,{tests:200,pass:200,fail:0,cancelled:0,skipped:0});assert.deepEqual(full.log,frozen.fullRegression.log);assert.equal(frozen.fullRegression.finalSourceFullRegressionPerformed,true);await tap(full.log,200);
 for(const p of frozen.evidenceArtifacts)await add(p);
 return{schemaVersion:1,scope:'Actual nested redgreen/current13 and full200 receipts, exact sourceBefore/sourceAfter, original4pass9fail red retained. Historical core187/wrapper28/adapter8 remain separate evidence.',exactSourceSha256:frozen.sourceSha256,fullRegressionPerformedOnFinalSource:true,focusedPassed:13,fullPassed:200,redOriginalPassed:4,redOriginalFailed:9,capacityAccepted:false,pins};
}
